"""F-14.10 measurement 1: does ReadDirectoryChangesW deliver a DISTINGUISHABLE,
drainable event for a DIRECTORY named-stream create+delete?

The ABA the tenth review left open is:

    dir has no stream :s      (initial inventory)
    create dir:s, write it    (during the interval)
    delete dir:s              (before the final inventory)
    dir has no stream :s      (final inventory == initial)

The inventory detector sees before == after. The lock cannot pre-open a
stream that does not exist. The write observer is the only thing that runs
FOR the whole interval, so the question is entirely: what does
ReadDirectoryChangesW put in the buffer when a directory's named stream
appears and disappears, and can it be told apart from an entry move
(which also arrives as `modified <dir>` and is deliberately forgiven)?

No production code is touched. This only measures the primitive.
"""
from __future__ import annotations

# The stream-specific notify filters and actions, which the current
# observer does NOT request and does NOT map.
NOTIFY = {
    "FILE_NAME": 0x001, "DIR_NAME": 0x002, "ATTRIBUTES": 0x004,
    "SIZE": 0x008, "LAST_WRITE": 0x010, "LAST_ACCESS": 0x020,
    "CREATION": 0x040, "SECURITY": 0x100,
    "STREAM_NAME": 0x200, "STREAM_SIZE": 0x400, "STREAM_WRITE": 0x800,
}
ACTIONS = {1: "added", 2: "removed", 3: "modified", 4: "renamed_from",
           5: "renamed_to", 6: "added_stream", 7: "removed_stream",
           8: "modified_stream", 9: "removed_by_delete"}

    _fields_ = (("Internal", ctypes.c_void_p), ("InternalHigh", ctypes.c_void_p),

k32.WaitForSingleObject.argtypes = [wintypes.HANDLE, wintypes.DWORD]
k32.CloseHandle.argtypes = [wintypes.HANDLE]
k32.CancelIo.argtypes = [wintypes.HANDLE]


def drain(root: Path, notify: int, action: "callable", label: str) -> list[str]:
    """Arm on `root`, run `action()`, drain every record until quiet."""
    handle = k32.CreateFileW(str(root), FILE_LIST_DIRECTORY, FILE_SHARE_ALL,
                             None, OPEN_EXISTING, BACKUP | OVERLAPPED, None)
    if handle == INVALID:
        return [f"watch failed: {ctypes.get_last_error()}"]
    buf = ctypes.create_string_buffer(64 * 1024)
    events: list[str] = []

    def arm() -> _Overlapped:
        ov = _Overlapped()
        ov.hEvent = k32.CreateEventW(None, True, False, None)
        returned = wintypes.DWORD(0)
        ok = k32.ReadDirectoryChangesW(handle, buf, len(buf), True, notify,
                                       ctypes.byref(returned), ctypes.byref(ov), None)
        if not ok:
            events.append(f"arm failed: {ctypes.get_last_error()}")
        return ov

    ov = arm()
    action()
    # Drain: each completed read is parsed, then re-armed, until a read
    # times out (nothing more is coming).
    for _ in range(20):
        if k32.WaitForSingleObject(ov.hEvent, 400) != 0:
            break
        got = wintypes.DWORD(0)
        if not k32.GetOverlappedResult(handle, ctypes.byref(ov), ctypes.byref(got), False):
            break
        if got.value == 0:
            # overflow: buffer too small, notifications lost
            events.append("OVERFLOW (buffer returned 0 bytes)")
            ov = arm()
            continue
        offset = 0
        while True:
            nxt = int.from_bytes(buf[offset:offset + 4], "little")
            act = int.from_bytes(buf[offset + 4:offset + 8], "little")
            length = int.from_bytes(buf[offset + 8:offset + 12], "little")
            name = bytes(buf[offset + 12:offset + 12 + length]).decode("utf-16-le")
            events.append(f"{ACTIONS.get(act, f'action-{act}')}: {name}")
            if not nxt:
                break
            offset += nxt
        ov = arm()
    k32.CancelIo(handle)
    k32.CloseHandle(handle)
    return events

"""F-14.10 measurement 2: the USN change journal as an ABA-proof primitive,
and its preconditions in THIS process's privilege context.

The USN journal is append-only and monotonic: it records that a change
happened even when the change is later reverted, which is exactly the
version-tag that defeats A->B->A. But it has hard preconditions, and the
whole point of this unit is to find out which of them hold WITHOUT
assuming: is the journal enabled, can THIS process (unelevated) read it,
does a directory-stream create+delete leave a STREAM_CHANGE record, and
can journal deletion/purge be detected.

Read-only against the real volume except for a stream ABA on a temp file
and directory. No production code touched.
"""

USN_REASON = {
    0x00000001: "DATA_OVERWRITE", 0x00000002: "DATA_EXTEND",
    0x00000004: "DATA_TRUNCATION", 0x00000010: "NAMED_DATA_OVERWRITE",
    0x00000020: "NAMED_DATA_EXTEND", 0x00000040: "NAMED_DATA_TRUNCATION",

    0x00400000: "TRANSACTED_CHANGE", 0x80000000: "CLOSE",
}

def open_volume(drive: str) -> tuple[int | None, int]:
    handle = k32.CreateFileW(f"\\\\.\\{drive}", GENERIC_READ, FILE_SHARE_ALL,
                             None, OPEN_EXISTING, 0, None)
    if handle == INVALID:
        return None, ctypes.get_last_error()
    return handle, 0


def query_journal(handle: int) -> tuple[dict | None, int]:
    out = ctypes.create_string_buffer(80)
    returned = wintypes.DWORD(0)
    ok = k32.DeviceIoControl(handle, FSCTL_QUERY_USN_JOURNAL, None, 0,
                             out, len(out), ctypes.byref(returned), None)
    if not ok:
        return None, ctypes.get_last_error()
    # USN_JOURNAL_DATA_V0: journalID, firstUsn, nextUsn, lowestValidUsn,
    # maxUsn, maximumSize, allocationDelta (all 8 bytes except last two)
    jid, first, nxt, lowest, maxusn, maxsize, alloc = struct.unpack(
        "<QqqqqQQ", out.raw[:56])
    return {"journal_id": jid, "first_usn": first, "next_usn": nxt,
            "lowest_valid": lowest, "max_usn": maxusn,
            "max_size": maxsize, "alloc_delta": alloc}, 0


def read_journal(handle: int, start_usn: int, jid: int) -> tuple[list, int, int]:
    """Read records from start_usn forward. Returns (records, next, err)."""
    # READ_USN_JOURNAL_DATA_V0: StartUsn q, ReasonMask I, ReturnOnlyOnClose I,
    # Timeout Q, BytesToWaitFor Q, UsnJournalID Q = 40 bytes.
    read_in = struct.pack("<qIIQQQ", start_usn, 0xFFFFFFFF, 0, 0, 0, jid)
    buf = ctypes.create_string_buffer(64 * 1024)
    returned = wintypes.DWORD(0)
    records: list[tuple[int, int, int]] = []  # (frn, parent_frn, reason)
    ok = k32.DeviceIoControl(handle, FSCTL_READ_USN_JOURNAL,
                             read_in, len(read_in), buf, len(buf),
                             ctypes.byref(returned), None)
    if not ok:
        return [], 0, ctypes.get_last_error()
    if returned.value < 8:
        return [], start_usn, 0
    next_usn = struct.unpack("<q", buf.raw[:8])[0]
    offset = 8
    data = buf.raw[:returned.value]
    while offset < returned.value:
        rec_len = struct.unpack("<I", data[offset:offset + 4])[0]
        if rec_len == 0:
            break
        # USN_RECORD_V2 layout
        major = struct.unpack("<H", data[offset + 4:offset + 6])[0]
        frn = struct.unpack("<Q", data[offset + 8:offset + 16])[0]
        parent = struct.unpack("<Q", data[offset + 16:offset + 24])[0]
        reason = struct.unpack("<I", data[offset + 40:offset + 44])[0]
        name_len = struct.unpack("<H", data[offset + 56:offset + 58])[0]
        name_off = struct.unpack("<H", data[offset + 58:offset + 60])[0]
        name = data[offset + name_off:offset + name_off + name_len].decode(
            "utf-16-le", "replace")
        records.append((frn, parent, reason, name, major))
        offset += rec_len
    return records, next_usn, 0


def reasons(mask: int) -> str:
    return "|".join(v for k, v in USN_REASON.items() if mask & k) or "none"
