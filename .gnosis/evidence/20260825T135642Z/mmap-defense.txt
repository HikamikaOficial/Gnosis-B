def opn(path, share, access=GR | GW):
    h = k32.CreateFileW(path, access, share, None, OPEN_EXISTING, NORMAL, None)
    if h == INVALID:
        raise OSError(ctypes.get_last_error(), f"open {path}")
    return h


def mapping(path, share):
    fh = opn(path, share)
    mp = k32.CreateFileMappingW(fh, None, PAGE_RW, 0, 0, None)
    if not mp:
        e = ctypes.get_last_error(); k32.CloseHandle(fh)
        raise OSError(e, "CreateFileMapping")
    view = k32.MapViewOfFile(mp, MAP_WRITE, 0, 0, 0)
    if not view:
        e = ctypes.get_last_error(); k32.CloseHandle(mp); k32.CloseHandle(fh)
        raise OSError(e, "MapViewOfFile")
    return fh, mp, view


def teardown(fh=None, mp=None, view=None):
    if view:
        k32.UnmapViewOfFile(view)
    if mp:
        k32.CloseHandle(mp)
    if fh:
        k32.CloseHandle(fh)

def lock_enforced(repo: Path):
    lock = create_input_lock(repo)
    outcome = lock.acquire(sorted(covered_paths(repo)))
    lock.release()
    return outcome
