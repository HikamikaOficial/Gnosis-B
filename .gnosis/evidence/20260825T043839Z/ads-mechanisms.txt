class WIN32_FIND_STREAM_DATA(ctypes.Structure):
    _fields_ = (("StreamSize", ctypes.c_longlong),

k32.FindFirstStreamW.restype = wintypes.HANDLE

def streams_of(path: Path) -> list[tuple[str, int]]:
    """FindFirstStreamW/FindNextStreamW: names come back as :name:$DATA."""
    data = WIN32_FIND_STREAM_DATA()
    handle = k32.FindFirstStreamW(str(path), 0, ctypes.byref(data), 0)
    if handle == INVALID_HANDLE_VALUE:
        err = ctypes.get_last_error()
        return [("<FindFirstStreamW failed %d>" % err, -1)]
    found = []
    try:
        while True:
            found.append((data.cStreamName, data.StreamSize))
            if not k32.FindNextStreamW(handle, ctypes.byref(data)):
                if ctypes.get_last_error() != ERROR_HANDLE_EOF:
                    found.append(("<FindNextStreamW failed %d>"
                                  % ctypes.get_last_error(), -1))
                break
    finally:
        k32.FindClose(handle)
    return found


def hold(path: str, share: int, flags: int = 0) -> int | None:

                             flags, None)
    if handle == INVALID_HANDLE_VALUE:
        return None
    return handle

class _Overlapped(ctypes.Structure):
    _fields_ = (("Internal", ctypes.c_void_p), ("InternalHigh", ctypes.c_void_p),
                ("Offset", wintypes.DWORD), ("OffsetHigh", wintypes.DWORD),
                ("hEvent", wintypes.HANDLE))


def watch(root: Path, notify: int, target: Path) -> str:
    """Arm a watcher, write a named stream, report what came back."""
    k32.ReadDirectoryChangesW.argtypes = [
        wintypes.HANDLE, ctypes.c_void_p, wintypes.DWORD, wintypes.BOOL,
        wintypes.DWORD, wintypes.LPDWORD, ctypes.c_void_p, ctypes.c_void_p]
    k32.CreateEventW.restype = wintypes.HANDLE
    k32.WaitForSingleObject.argtypes = [wintypes.HANDLE, wintypes.DWORD]
    k32.GetOverlappedResult.argtypes = [wintypes.HANDLE, ctypes.c_void_p,
                                        wintypes.LPDWORD, wintypes.BOOL]

    handle = k32.CreateFileW(str(root), FILE_LIST_DIRECTORY, FILE_SHARE_ALL,
                             None, OPEN_EXISTING, FILE_FLAG_BACKUP_SEMANTICS
                             | 0x40000000, None)  # FILE_FLAG_OVERLAPPED
    if handle == INVALID_HANDLE_VALUE:
        return f"could not watch: {ctypes.get_last_error()}"
    buf = ctypes.create_string_buffer(64 * 1024)
    ov = _Overlapped()
    ov.hEvent = k32.CreateEventW(None, True, False, None)
    returned = wintypes.DWORD(0)
    ok = k32.ReadDirectoryChangesW(handle, buf, len(buf), True, notify,
                                   ctypes.byref(returned), ctypes.byref(ov), None)
    if not ok:
        k32.CloseHandle(handle)
        return f"arm failed: {ctypes.get_last_error()}"

    def writer() -> None:
        time.sleep(0.15)
        with open(f"{target}:watched-stream", "w", encoding="utf-8") as fh:
            fh.write("WRITTEN DURING THE INTERVAL")

    thread = threading.Thread(target=writer)
    thread.start()
    thread.join()
    time.sleep(0.3)

    events: list[str] = []
    if k32.WaitForSingleObject(ov.hEvent, 1500) == 0:
        got = wintypes.DWORD(0)
        if k32.GetOverlappedResult(handle, ctypes.byref(ov), ctypes.byref(got), False):
            offset = 0
            while True:
                nxt = int.from_bytes(buf[offset:offset + 4], "little")
                action = int.from_bytes(buf[offset + 4:offset + 8], "little")
                length = int.from_bytes(buf[offset + 8:offset + 12], "little")
                name = bytes(buf[offset + 12:offset + 12 + length]).decode("utf-16-le")
                events.append(f"action={action} name={name!r}")
                if not nxt:
                    break
                offset += nxt
    k32.CloseHandle(handle)
    return "; ".join(events) if events else "NOTHING DELIVERED"

def named_streams(path: Path) -> tuple[list[tuple[str, int]], int]:
    """(named streams, error). `::$DATA` is the main stream and is skipped."""
    data = FIND_STREAM_DATA()
    handle = k32.FindFirstStreamW(str(path), 0, ctypes.byref(data), 0)
    if handle == INVALID_HANDLE_VALUE:
        err = ctypes.get_last_error()
        return [], 0 if err == ERROR_HANDLE_EOF else err
    found: list[tuple[str, int]] = []
    try:
        while True:
            name = data.cStreamName
            if name != "::$DATA":
                found.append((name, data.StreamSize))
            if not k32.FindNextStreamW(handle, ctypes.byref(data)):
                err = ctypes.get_last_error()
                return found, 0 if err == ERROR_HANDLE_EOF else err
    finally:
        k32.FindClose(handle)
