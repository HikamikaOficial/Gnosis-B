def git(repo: Path, *args: str) -> str:
    return subprocess.run(["git", *args], cwd=repo, capture_output=True,
                          text=True, check=True).stdout


def make_repo(root: Path) -> Path:
    repo = root / "repo"
    repo.mkdir()
    git(repo, "init")
    git(repo, "config", "user.email", "test@example.com")
    git(repo, "config", "user.name", "Test")

    git(repo, "add", "probe.txt")
    git(repo, "commit", "-m", "init")
    return repo


def write_stream(path: Path, stream: str, text: str) -> None:
    """A named stream is just a path with a colon in it."""
    with open(f"{path}:{stream}", "w", encoding="utf-8") as handle:
        handle.write(text)


def read_stream(path: Path, stream: str) -> str:
    with open(f"{path}:{stream}", encoding="utf-8") as handle:
        return handle.read()


def list_streams(path: Path) -> list[str]:
    """`dir /r` is the only shell way; PowerShell is the readable one."""
    out = subprocess.run(
        ["powershell", "-NoProfile", "-Command",
         f"Get-Item -LiteralPath '{path}' -Stream * | "
         "Select-Object -ExpandProperty Stream"],
        capture_output=True, text=True, errors="replace", check=False)
    return [line.strip() for line in out.stdout.splitlines() if line.strip()]

def capture(repo: Path, staging: Path) -> dict:
    result = run_capture(repo, [CHECK], staging)
    manifest = json.loads(
        (result.bundle / "input-manifest.json").read_text(encoding="utf-8"))
    protection = result.boundary.protection
    return {
        "check_tail": result.checks[0].tail if result.checks else "(no checks)",
        "identity_digest": protection.get("identity_digest"),
        "content_digest": protection.get("content_digest"),
        "locked": protection.get("locked_inputs"),
        "byte_bound": protection.get("byte_bound_inputs"),
        "fully_bound": protection.get("fully_bound"),
        "boundary": result.boundary.verdict.value,
        "violations": list(result.boundary.violations),
        "evidence_valid": result.evidence_valid,
        "exit_code": result.exit_code,
        "manifest_keys": sorted(manifest["inputs"]),
        "probe_digest": manifest["inputs"].get("probe.txt"),
        "ads_named": [k for k in manifest["inputs"] if ":" in k],
    }


def show(label: str, run: dict) -> None:

    print(f"  boundary                    : {run['boundary']}  {run['violations']}")
    print(f"  evidence_valid / exit       : {run['evidence_valid']} / {run['exit_code']}")
    print()
