===== mic_lib.py =====
"""Windows MIC authority primitives for F-17 OS AUTHORITY BOUNDARY (investigation).

Launch a child at MEDIUM integrity from an elevated (High) parent, using the
parent's LINKED (filtered, Medium) token via CreateProcessWithTokenW (needs
SeImpersonatePrivilege, which this environment has enabled). Also: read a
process token's integrity level, and set a directory's mandatory label.

No production code; scratchpad investigation only.
"""
from __future__ import annotations
import ctypes as C
from ctypes import wintypes as W

k32 = C.WinDLL("kernel32", use_last_error=True)
a32 = C.WinDLL("advapi32", use_last_error=True)

# Prototypes — without these, 64-bit HANDLEs (incl. the GetCurrentProcess
# pseudo-handle 0xFFFFFFFFFFFFFFFF) truncate to 32-bit and calls fail with
# ERROR_INVALID_HANDLE.
k32.GetCurrentProcess.restype = W.HANDLE
k32.GetCurrentProcess.argtypes = []
k32.WaitForSingleObject.argtypes = [W.HANDLE, W.DWORD]
k32.GetExitCodeProcess.argtypes = [W.HANDLE, C.POINTER(W.DWORD)]
k32.CloseHandle.argtypes = [W.HANDLE]
a32.OpenProcessToken.argtypes = [W.HANDLE, W.DWORD, C.POINTER(W.HANDLE)]
a32.OpenProcessToken.restype = W.BOOL
a32.GetTokenInformation.argtypes = [W.HANDLE, C.c_int, C.c_void_p, W.DWORD,
                                    C.POINTER(W.DWORD)]
a32.GetTokenInformation.restype = W.BOOL
a32.DuplicateTokenEx.argtypes = [W.HANDLE, W.DWORD, C.c_void_p, C.c_int,
                                 C.c_int, C.POINTER(W.HANDLE)]
a32.DuplicateTokenEx.restype = W.BOOL
a32.GetSidSubAuthorityCount.argtypes = [C.c_void_p]
a32.GetSidSubAuthority.argtypes = [C.c_void_p, W.DWORD]
a32.SetTokenInformation.argtypes = [W.HANDLE, C.c_int, C.c_void_p, W.DWORD]
a32.SetTokenInformation.restype = W.BOOL
a32.ConvertStringSidToSidW.argtypes = [W.LPCWSTR, C.POINTER(C.c_void_p)]
a32.ConvertStringSidToSidW.restype = W.BOOL
a32.GetLengthSid.argtypes = [C.c_void_p]
a32.GetLengthSid.restype = W.DWORD

TOKEN_ADJUST_DEFAULT = 0x0080
SE_GROUP_INTEGRITY = 0x00000020
SID_MEDIUM = "S-1-16-8192"
SID_LOW = "S-1-16-4096"

TOKEN_QUERY = 0x0008
TOKEN_DUPLICATE = 0x0002
TOKEN_ASSIGN_PRIMARY = 0x0001
TOKEN_ALL_ACCESS = 0xF01FF
TokenLinkedToken = 33
TokenIntegrityLevel = 25
SecurityImpersonation = 2
TokenPrimary = 1
LOGON_WITH_PROFILE = 0x00000001
CREATE_UNICODE_ENVIRONMENT = 0x00000400


class TOKEN_LINKED_TOKEN(C.Structure):
    _fields_ = [("LinkedToken", W.HANDLE)]


class SID_AND_ATTRIBUTES(C.Structure):
    _fields_ = [("Sid", C.c_void_p), ("Attributes", W.DWORD)]


class TOKEN_MANDATORY_LABEL(C.Structure):
    _fields_ = [("Label", SID_AND_ATTRIBUTES)]


class STARTUPINFO(C.Structure):
    _fields_ = [
        ("cb", W.DWORD), ("lpReserved", W.LPWSTR), ("lpDesktop", W.LPWSTR),
        ("lpTitle", W.LPWSTR), ("dwX", W.DWORD), ("dwY", W.DWORD),
        ("dwXSize", W.DWORD), ("dwYSize", W.DWORD), ("dwXCountChars", W.DWORD),
        ("dwYCountChars", W.DWORD), ("dwFillAttribute", W.DWORD),
        ("dwFlags", W.DWORD), ("wShowWindow", W.WORD), ("cbReserved2", W.WORD),
        ("lpReserved2", C.c_void_p), ("hStdInput", W.HANDLE),
        ("hStdOutput", W.HANDLE), ("hStdError", W.HANDLE),
    ]


class PROCESS_INFORMATION(C.Structure):
    _fields_ = [("hProcess", W.HANDLE), ("hThread", W.HANDLE),
                ("dwProcessId", W.DWORD), ("dwThreadId", W.DWORD)]


def _err(msg: str) -> None:
    raise OSError(f"{msg} (winerr {C.get_last_error()})")


def current_integrity() -> str:
    """Integrity level of the current process, as a name."""
    hToken = W.HANDLE()
    if not a32.OpenProcessToken(k32.GetCurrentProcess(), TOKEN_QUERY, C.byref(hToken)):
        _err("OpenProcessToken")
    size = W.DWORD()
    a32.GetTokenInformation(hToken, TokenIntegrityLevel, None, 0, C.byref(size))
    buf = C.create_string_buffer(size.value)
    if not a32.GetTokenInformation(hToken, TokenIntegrityLevel, buf, size, C.byref(size)):
        _err("GetTokenInformation(IL)")
    tml = C.cast(buf, C.POINTER(TOKEN_MANDATORY_LABEL)).contents
    # last sub-authority of the integrity SID is the RID
    get_count = a32.GetSidSubAuthorityCount
    get_count.restype = C.POINTER(C.c_ubyte)
    get_sub = a32.GetSidSubAuthority
    get_sub.restype = C.POINTER(W.DWORD)
    n = get_count(tml.Label.Sid)[0]
    rid = get_sub(tml.Label.Sid, n - 1)[0]
    names = {0x0000: "Untrusted", 0x1000: "Low", 0x2000: "Medium",
             0x3000: "High", 0x4000: "System"}
    return names.get(rid, f"0x{rid:04x}")


def lowered_primary_token(sid_string: str) -> W.HANDLE:
    """Duplicate the current token to PRIMARY and lower its integrity to
    `sid_string` (Medium or Low). Lowering one's own token IL needs no
    privilege; a lowered token can never be raised again."""
    hToken = W.HANDLE()
    if not a32.OpenProcessToken(
            k32.GetCurrentProcess(),
            TOKEN_QUERY | TOKEN_DUPLICATE | TOKEN_ASSIGN_PRIMARY | TOKEN_ADJUST_DEFAULT,
            C.byref(hToken)):
        _err("OpenProcessToken")
    primary = W.HANDLE()
    if not a32.DuplicateTokenEx(hToken, TOKEN_ALL_ACCESS, None,
                                SecurityImpersonation, TokenPrimary, C.byref(primary)):
        _err("DuplicateTokenEx")
    pSid = C.c_void_p()
    if not a32.ConvertStringSidToSidW(sid_string, C.byref(pSid)):
        _err("ConvertStringSidToSidW")
    label = TOKEN_MANDATORY_LABEL()
    label.Label.Sid = pSid
    label.Label.Attributes = SE_GROUP_INTEGRITY
    size = C.sizeof(SID_AND_ATTRIBUTES) + a32.GetLengthSid(pSid)
    if not a32.SetTokenInformation(primary, TokenIntegrityLevel, C.byref(label), size):
        _err("SetTokenInformation(IL)")
    return primary


def run_as_medium(cmdline: str, cwd: str, sid_string: str = SID_MEDIUM) -> int:
    """Launch cmdline at the given integrity (Medium by default). Exit code."""
    token = lowered_primary_token(sid_string)
    si = STARTUPINFO(); si.cb = C.sizeof(si)
    pi = PROCESS_INFORMATION()
    a32.CreateProcessWithTokenW.argtypes = [
        W.HANDLE, W.DWORD, W.LPCWSTR, W.LPWSTR, W.DWORD, C.c_void_p,
        W.LPCWSTR, C.POINTER(STARTUPINFO), C.POINTER(PROCESS_INFORMATION)]
    ok = a32.CreateProcessWithTokenW(
        token, LOGON_WITH_PROFILE, None, cmdline, CREATE_UNICODE_ENVIRONMENT,
        None, cwd, C.byref(si), C.byref(pi))
    if not ok:
        _err("CreateProcessWithTokenW")
    k32.WaitForSingleObject(pi.hProcess, 0xFFFFFFFF)
    code = W.DWORD()
    k32.GetExitCodeProcess(pi.hProcess, C.byref(code))
    k32.CloseHandle(pi.hProcess); k32.CloseHandle(pi.hThread)
    return code.value


if __name__ == "__main__":
    print("current integrity:", current_integrity())

def rec(name, fn, expect_denied):
    try:
        fn()
        verdict = "ALLOWED"
    except Exception as e:
        verdict = f"DENIED ({type(e).__name__}: {str(e).splitlines()[0][:70]})"
    ok = ("DENIED" in verdict) == expect_denied
    lines.append(f"{'OK ' if ok else 'XX '}{name}: {verdict}  [expect {'DENIED' if expect_denied else 'ALLOWED'}]")


def icacls_denied(args):
    r = subprocess.run(["icacls", anchor, *args], capture_output=True, text=True)
    if r.returncode != 0:
        raise PermissionError((r.stdout + r.stderr).strip().splitlines()[-1][:70])


def shell_write_anchor():
    tgt = os.path.join(anchor, "gc_shell.txt")
    subprocess.run(["cmd", "/c", f'echo gc> "{tgt}"'], capture_output=True, text=True)
    if not os.path.exists(tgt):
        raise PermissionError("cmd redirect to anchor produced no file (denied)")


def py_child_write_anchor():
    tgt = os.path.join(anchor, "gc_py.txt")
    code = ("import sys\n"
            "try:\n"
            f"    open(r'{tgt}','w').write('x'); print('WROTE')\n"
            "except Exception as e:\n"
            "    print('DENIED', type(e).__name__)\n")
    r = subprocess.run([sys.executable, "-c", code], capture_output=True, text=True)
    if "WROTE" in r.stdout or os.path.exists(tgt):
        raise RuntimeError("PRIVILEGE RECOVERY: python grandchild wrote the anchor")
    # denied is the expected/healthy path -> raise so rec() records DENIED
    raise PermissionError("python grandchild denied: " + r.stdout.strip()[:50])

def icacls(*args):
    return subprocess.run(["icacls", *args], capture_output=True, text=True)


def main():

    if os.path.isdir(BASE):
